import re

def clean_mermaid_code(mermaid_code):
    def clean_label(label):
        # Remove or replace unsafe characters
        label = re.sub(r'\(.*?\)', '', label)  # Remove parentheses and contents
        label = re.sub(r'[,:\"\'/]', '', label)  # Remove other unsafe characters
        return label.strip()

    cleaned_lines = []
    for line in mermaid_code.splitlines():
        match = re.match(r'(\w+)\[(.+?)\]', line)
        if match:
            node_id, label = match.groups()
            safe_label = clean_label(label)
            line = line.replace(f"[{label}]", f"[{safe_label}]")
        cleaned_lines.append(line)
    return '\n'.join(cleaned_lines)
